\begin{frame}{Three decisions, one athlete}
A competitive triathlete must decide:
\begin{enumerate}
  \item<1-> \textbf{How to train} --- hours per sport per intensity, over 12--20 weeks\\
        {\small fitness accrues slowly, fatigue quickly $\Rightarrow$ a dynamic trade-off}
  \item<2-> \textbf{Which races to enter} --- a season calendar under money,
        time-off and recovery limits
  \item<3-> \textbf{How to race} --- distribute a finite energy reserve over
        swim, bike, run
\end{enumerate}

\begin{frame}{Model 1 --- training-load allocation (LP)}
\textbf{Variables:} hours $h_{dbw} \ge 0$ per discipline $\times$ bucket $\times$ week
($3{\times}3{\times}16 = 144$)
\vspace{4pt}

\textbf{Banister dynamics as linear equalities} (exact --- smoothing is linear in load):
\[
L_w = \sum_{d,b} r_{db} h_{dbw}, \qquad
\text{CTL}_w = \lambda_c \text{CTL}_{w-1} + (1{-}\lambda_c)\tfrac{L_w}{7}, \qquad
\text{ATL}_w = \lambda_a \text{ATL}_{w-1} + (1{-}\lambda_a)\tfrac{L_w}{7}
\]

\textbf{Objective:} race-day performance
$\;\max\, p_T = k_1 \text{CTL}_T - k_2 \text{ATL}_T$
\hfill {\small (TSB-max alone would be degenerate: optimum = rest!)}
\vspace{4pt}

\textbf{Constraints:} weekly hour ceilings (phase-dependent) $\cdot$ discipline
min/max $\cdot$ 80/20 polarization $\cdot$ $\text{ATL} \le 110$ $\cdot$
ramp $\le +10\%$/week $\cdot$ taper caps
\end{frame}

\begin{frame}{Model 2 --- race-calendar selection (0--1 MIP)}
\textbf{17 candidate races} (Greek / European / long-haul), binary $x_i$
\[
\max \sum_i \underbrace{q_i(\alpha\,\text{pts}_i + \beta\,\text{strat}_i
+ \gamma\,\text{enj}_i + \varphi\,\text{fit}_i)}_{\text{weather-risk-adjusted value}} x_i
\]
\begin{itemize}
  \item \textbf{Two-dimensional knapsack:} $\sum \kappa_i x_i \le 2500$ EUR
        \;and\; $\sum \delta_i x_i \le 12$ vacation days
  \item recovery spacing $x_i + x_j \le 1$ (1/2/4 weeks by distance class);
        $\le 2$ races/month; $\ge 1$ A-race
  \item \textbf{readiness gate from Model 1:} projected CTL must reach the class
        requirement --- excludes a high-value early-season 70.3
  \item suitability $\text{fit}_i$ from \emph{objective} course data:
        elevation gain/km vs.\ athlete's flat-course preference
\end{itemize}
\end{frame}

\begin{frame}{Model 2 --- the dual problem}
LP relaxation is already in standard form
$\max\{c^{T}x \mid Ax \le b,\, x \ge 0\}$, so the dual is
$\min\{b^{T}y \mid A^{T}y \ge c,\, y \ge 0\}$:
\[
\kappa_i y_\mathcal{B} + \delta_i y_\mathcal{D} + y_{\text{month}(i)}
+ \sum_{j} y_{ij} - [i \in A] y_A \;\ge\; v_i
\]
\begin{center}\small
\emph{a race enters only if its value covers the shadow cost of what it consumes}
\end{center}
\begin{center}\small
\begin{tabular}{llr}
\toprule
Dual & Meaning & Price\\
\midrule
$y_{13,14}$ & Hellenic Champs vs.\ Costa Navarino conflict & \textbf{56.40}\\
$y_{06,07}$ & Hamburg vs.\ Thessaloniki conflict & 31.67\\
$y_\mathcal{B}$ & one extra euro & 0.094\\
$y_\mathcal{D}$ & one extra vacation day & $\approx 0$\\
\bottomrule
\end{tabular}
\end{center}
\vspace{2pt}
$\Rightarrow$ the scarce resource is neither money nor days off but
\textbf{calendar space}: recovery between races.
\end{frame}

\begin{frame}{Model 2 --- network-flow extension}
Drop the knapsacks and monthly cap $\Rightarrow$ only recovery spacing remains:
\begin{center}
\textbf{feasible calendars = paths in a DAG} \quad
(nodes: races in week order; arc $i \to j$ iff gap $\ge$ recovery$(i)$)
\end{center}
\vspace{4pt}
Three methods, identical result ($z = 639.4$):
\begin{enumerate}
  \item dynamic programming (longest path, topological order)
  \item \textbf{flow LP with continuous variables} --- solution integral anyway:
        network matrix $\Rightarrow$ \emph{totally unimodular}
  \item restricted 0--1 MIP
\end{enumerate}
\vspace{2pt}
{\small Node--arc incidence matrix $\Rightarrow$ every basis is a spanning tree:
exactly the class \emph{Network Simplex} is designed for (18 nodes, 136 arcs).}
\vspace{4pt}
$\Rightarrow$ the knapsack constraints are precisely what \emph{breaks} the network
structure and forces Branch \& Bound in the full model.
\end{frame}

\begin{frame}{Model 3 --- race pacing (LP with linearization)}
\textbf{Variables:} minutes $t_{\ell z}$ per leg $\times$ intensity zone (Z1--Z5);
15 variables
\vspace{4pt}

\textbf{Linearization:} within a zone, speed and energy rate are constants ---
bike speed $\propto$ (power)$^{1/3}$ (aero drag), economy degrades with intensity
\vspace{4pt}
\[
\min \sum_{\ell,z} t_{\ell z} + T_1 + T_2
\quad \text{s.t.} \quad
\underbrace{\textstyle\sum_z s_{\ell z} t_{\ell z} \ge d_\ell}_{\text{distances}},\;
\underbrace{\textstyle\sum e_{\ell z} t_{\ell z} \le k_E \cdot \text{CTL}_T}_{\text{energy budget (Model 1)}},\;
\underbrace{\text{Z4+ time caps}}_{\text{per leg}},\;
\underbrace{\varphi \cdot G}_{\text{bike} \to \text{run}}
\]
The coupling: each kJ of hard bike work reduces effective run coverage.
\end{frame}

\begin{itemize}
  \item ``35 km/h at FTP'' describes a \emph{flat} course; scale it by

  \item fixes both hilly courses almost exactly --- but \textbf{over-corrects}

  \item so: shipped \textbf{disabled}. Three races can \emph{locate} a
        systematic error, not \emph{calibrate} it away
\end{itemize}

\begin{frame}{Integrating the pipeline: coherence + fixed point}
\textbf{Do the objectives agree?} Model 1 maximizes
$p_T = k_1\text{CTL}_T - k_2\text{ATL}_T$; the athlete wants \emph{race time}.
\begin{itemize}\itemsep1pt
  \item default budget $E = k_E\text{CTL}$ ignores ATL $\Rightarrow$ four
        different plans all give \textbf{the same 140.2 min} --- Model 3 is blind
        to training
  \item form-based $E = E_{\text{base}} + k_E(\text{CTL}_T - \lambda\text{ATL}_T)$
        is affine in $p_T$ \textbf{iff} $\lambda = k_2/k_1$
        $\Rightarrow \arg\max p_T = \arg\min$ time
  \item $E>0$ needs $\lambda < \text{CTL}_T/\text{ATL}_T = 1.75$, so the assumed
        $k_2 = 2$ is \emph{incoherent}; cost ranging says the plan is invariant on
        $[1.29, 2.95]$, so $\lambda = 1.5$ fixes it \textbf{changing no variable}
\end{itemize}
\vspace{2pt}
\textbf{Which comes first?} The A-race sets Model 1's macrocycle; Model 1's load
drives Model 2's readiness gate. Iterate instead of assuming:
\begin{center}\small

\toprule

\end{frame}

\begin{frame}{What the models actually recommend}
\begin{itemize}
  \item \textbf{Train more, much easier.} Actual: 6.6 h/wk at \textbf{37\%}
        hard. Optimal: 9.0 h/wk at \textbf{15\%}. Anti-polarised --- the one
        change with the largest payoff.
  \item \textbf{Buying training time is worthless} above 10.5 h/wk (zero shadow
        price); \emph{recovery capacity} is what binds.
  \item \textbf{Build endurance, not watts.} $+1$ CTL $=$ 69--84 s;
        $+1$ W $\approx 6$ s below 260 W, $\approx 0$ above.
  \item \textbf{Ride Z3, run hard, pace evenly.} The 70.3 positive split
        (4:53 $\to$ 5:52/km, HR 162 $\to$ 145) cost $\approx$ 29 min.
  \item \textbf{Fuel 75 g/h}; gut training worth $\approx 1.7$ min per kJ/min
        --- at 70.3 only.
  \item \textbf{Protect the calendar, not the budget.} Recovery conflict dual
        \textbf{56.4} vs.\ 0.094 per euro.
  \item \textbf{Skip early-season Dubai} (not fit in time); if the A-race is a
        70.3, build \textbf{20 weeks}, not 16.
\end{itemize}
\end{frame}

\begin{frame}{Backup: limitations --- stated honestly}
\begin{itemize}
  \item \textbf{Weekly time step} for Banister dynamics (daily would shift constants,
        not structure)
  \item \textbf{Sequential pipeline}, not joint optimization --- co-optimizing
        training + calendar is a natural (larger) MIP extension
  \item \textbf{Constructed parameters} in Models 2--3 (race values, zone speeds,
        $k_E$) --- documented assumptions; sensitivity analysis shows which
        conclusions survive perturbation
  \item LP's $+35$ TSB over-taper traced to race-week minimum constraints,
        not to the model concept
\end{itemize}
\end{frame}
